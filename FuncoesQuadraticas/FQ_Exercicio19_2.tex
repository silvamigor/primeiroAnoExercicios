\definevalorquestao

\begin{exercicioBanco}[\valorquestao]
A função afim \(f\) e a função quadrática \(g\) estão representadas a seguir.
%
\begin{center}
    \begin{tikzpicture}[scale=0.6]
        % Eixos
        \draw[->] (-3.5,0) -- (4,0) node[right] {\(x\)};
        \draw[->] (0,-1.5) -- (0,5.5) node[above] {\(y\)};
              
        % Gráfico
        \draw[domain=-2:3.3, smooth, variable=\x, blue, thick] plot ({\x}, {-(2/3)*\x + 2});
        \node[blue, above left] at (-2, {-(-2) + 2}) {\(f\)};
        \draw[domain=-1.2:3.2, smooth, variable=\x, red, thick] plot ({\x}, {-\x*\x + 2*\x + 3}) node[left, yshift=80pt] {\(g\)};
              
        % Pontos escolhidos
        \draw[thick] (-1,-0.2) -- (-1,0.2) node[above left]{\small\(-1\)};
        \draw[thick] (3,-0.2) -- (3,0.2) node[above right]{\small\(3\)};
        \draw[thick] (-0.2,3) -- (0.2,3) node[right]{\small\(3\)};
        \draw[thick] (-0.2,2) -- (0.2,2) node[right]{\small\(2\)};
    \end{tikzpicture}
\end{center}
%
Determine o valor de \(x < 0\) tal que \(f(x) = g(x)\).

% Define as alternativas
\newcommand{\alternativas}{%
    \begin{center}
        \begin{tabularx}{\textwidth}{XXXXX}

            (d) \(-\dfrac{1}{2}\). &
            (e) \(-\dfrac{1}{3}\).
        \end{tabularx}
    \end{center}
}

\newcommand{\resposta}{E}

% Lógica condicional para exibição

    }{%

    }
}
\end{exercicioBanco}
